# call word ptr cs:[bx-0x2BC6] bei 0000:D2DA
# Tabellenbasis (0 - 0x2BC6) & 0xFFFF = 0xD43A, 7 Einträge.
# Index kommt aus `and bl,0x0F / shl bl,1`.
# Alle Handler prüfen cs:[0xD5C8] gegen 4 -> Grafikadapter-Klasse.
# Das ist der Pixelformat-Dispatcher (CGA/Tandy/EGA/VGA), kein Entpacker.

0000:D448   # Eintrag 1 - cx=0x2000 (8 KB Plane)

0000:D4C7   # Eintrag 4
0000:D4F2   # Eintrag 5
0000:D525   # Eintrag 6 - Kette: D448, D456, D448, D496

0000:2916   # Bit 6
0000:29B8   # Bit 7 - int 15h AX=0xC200, PS/2-Maus
